### 1.4 هيكل النموذج (Form)

النظام يستخدم `useState` + `useTransition` + `fetch()` — **لا يستخدم react-hook-form**:

```tsx
"use client";

import { useRouter } from "next/navigation";
import { useState, useTransition } from "react";

export function EditForm({ entityId, initialValues, onSuccess }: Props) {
  const router = useRouter();
  const [error, setError] = useState<string | null>(null);
  const [isPending, startTransition] = useTransition();
  const [formState, setFormState] = useState(initialValues);

  function updateField<K extends keyof typeof formState>(key: K, value: typeof formState[K]) {
    setFormState((current) => ({ ...current, [key]: value }));
  }

  function handleSubmit(event: React.FormEvent) {
    event.preventDefault();
    startTransition(async () => {
      try {
        setError(null);
        const response = await fetch(`/api/feature/${entityId}`, {
          method: "PATCH",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify(formState),
        });
        const data = await response.json();
        if (!response.ok) throw new Error(data.error?.message ?? "حدث خطأ");
        onSuccess?.();
        router.refresh();
      } catch (err) {
        setError(err instanceof Error ? err.message : "حدث خطأ غير متوقع");
      }
    });
  }

  return (
    <form onSubmit={handleSubmit}>
      <div className="space-y-5 rounded-[1.5rem] border border-border/70 bg-white/80 p-5">
        {/* form fields */}
        {error && <div className="rounded-2xl border border-rose-200 bg-rose-50 px-4 py-3 text-sm text-rose-700">{error}</div>}
        <div className="flex flex-wrap gap-3">
          <button className="rounded-2xl bg-primary px-5 py-3 text-sm font-semibold text-primary-foreground disabled:opacity-60" disabled={isPending} type="submit">
            حفظ
          </button>
        </div>
      </div>
    </form>
  );
}
```

### 3.2 إنشاء خدمة تحديث العميل

**ملف جديد**: `src/server/services/customers-service.ts`

```typescript
import { z } from "zod";
import type { SessionUser } from "@/lib/auth/get-session-user";
import { requirePermission } from "@/lib/auth/permissions";
import { notFound } from "@/lib/errors/api-error";
import { apiErrorResponse, toApiError, invalidRequest } from "@/lib/errors/api-error";
import { createAdminSupabaseClient } from "@/lib/supabase/admin";
import { updateCustomerSchema, type UpdateCustomerInput } from "@/features/customers/schemas/customer-form";

export async function updateCustomer(input: {
  customerId: string;
  payload: UpdateCustomerInput;
  sessionUser: SessionUser;
}) {
  requirePermission(input.sessionUser, "customers.read"); // أي مستخدم يقدر يقرأ يقدر يعدل ملاحظات - أو أنشئ صلاحية جديدة
  const parsed = updateCustomerSchema.parse(input.payload);
  const client = createAdminSupabaseClient();

  const { data: existing } = await client.from("customers").select("*").eq("id", input.customerId).maybeSingle();
  if (!existing) throw notFound("تعذر العثور على العميل المطلوب");

  const updatePayload: Record<string, unknown> = {};
  if (parsed.customerName !== undefined) updatePayload.customer_name = parsed.customerName;
  if (parsed.mobile !== undefined) updatePayload.mobile = parsed.mobile;
  if (parsed.email !== undefined) updatePayload.email = parsed.email;
  if (parsed.nationalId !== undefined) updatePayload.national_id = parsed.nationalId;
  if (parsed.notes !== undefined) updatePayload.notes = parsed.notes;

  if (Object.keys(updatePayload).length === 0) {
    return existing;
  }

  const { data, error } = await client
    .from("customers")
    .update(updatePayload)
    .eq("id", input.customerId)
    .select("*")
    .single();

  if (error || !data) throw new Error(error?.message ?? "تعذر تحديث بيانات العميل");
  return data;
}

export function handleCustomerServiceError(error: unknown): Response {
  if (error instanceof z.ZodError) {
    return apiErrorResponse(invalidRequest(error.issues[0]?.message ?? "بيانات غير صحيحة", error.flatten()));
  }
  return apiErrorResponse(toApiError(error));
}
```

### 5.2 إنشاء خدمة تحديث القسط

**ملف جديد**: `src/server/services/installments-service.ts`

```typescript
import { z } from "zod";
import type { SessionUser } from "@/lib/auth/get-session-user";
import { requirePermission } from "@/lib/auth/permissions";
import { notFound, apiErrorResponse, toApiError, invalidRequest } from "@/lib/errors/api-error";
import { createAdminSupabaseClient } from "@/lib/supabase/admin";
import { getEgyptToday } from "@/lib/dates/egypt";
import { updateInstallmentSchema, type UpdateInstallmentInput } from "@/features/installments/schemas/installment-form";

export async function updateInstallment(input: {
  installmentId: string;
  payload: UpdateInstallmentInput;
  sessionUser: SessionUser;
}) {
  requirePermission(input.sessionUser, "installments.read");
  const parsed = updateInstallmentSchema.parse(input.payload);
  const client = createAdminSupabaseClient();

  const { data: existing } = await client.from("installments").select("*").eq("id", input.installmentId).maybeSingle();
  if (!existing) throw notFound("تعذر العثور على القسط المطلوب");

  const amountCollected = parsed.amountCollected ?? existing.amount_collected;
  const amountDue = existing.amount_due;
  const amountOutstanding = Math.max(amountDue - amountCollected, 0);
  const today = getEgyptToday();

  // اشتقاق حالة الدفع
  let paymentStatus: string;
  if (amountOutstanding <= 0) {
    paymentStatus = "paid";
  } else if (amountCollected > 0) {
    paymentStatus = "partial";
  } else if (existing.due_date < today) {
    paymentStatus = "overdue";
  } else {
    paymentStatus = "unpaid";
  }

  // اشتقاق أيام التأخير
  const delayDays = amountOutstanding > 0 && existing.due_date < today
    ? Math.floor((new Date(today).getTime() - new Date(existing.due_date).getTime()) / 86400000)
    : 0;

  // اشتقاق bucket التأخير
  let delayBucket = "not_due";
  if (delayDays > 90) delayBucket = "90_plus";
  else if (delayDays > 60) delayBucket = "61_90";
  else if (delayDays > 30) delayBucket = "31_60";
  else if (delayDays > 0) delayBucket = "1_30";

  const { data, error } = await client
    .from("installments")
    .update({
      amount_collected: amountCollected,
      amount_outstanding: amountOutstanding,
      payment_status: paymentStatus,
      payment_date: parsed.paymentDate ?? existing.payment_date,
      receipt_reference: parsed.receiptReference ?? existing.receipt_reference,
      penalty_amount: parsed.penaltyAmount ?? existing.penalty_amount,
      delay_days: delayDays,
      delay_bucket: delayBucket,
    })
    .eq("id", input.installmentId)
    .select("*")
    .single();

  if (error || !data) throw new Error(error?.message ?? "تعذر تحديث القسط");
  return data;
}

export function handleInstallmentServiceError(error: unknown): Response {
  if (error instanceof z.ZodError) {
    return apiErrorResponse(invalidRequest(error.issues[0]?.message ?? "بيانات غير صحيحة", error.flatten()));
  }
  return apiErrorResponse(toApiError(error));
}
```

### 7.1 استعلام بيانات الرسوم

**ملف جديد**: `src/server/queries/dashboard/get-dashboard-charts.ts`

```typescript
import type { SessionUser } from "@/lib/auth/get-session-user";
import { requirePermission } from "@/lib/auth/permissions";
import {
  buildInstallmentsByContract,
  buildProjectById,
  filterContractsByScope,
  loadReadModelData,
} from "@/server/queries/read-model-helpers";

export type CollectionByProjectItem = {
  projectName: string;
  totalDue: number;
  totalCollected: number;
  totalOutstanding: number;
};

export type AgingDistributionItem = {
  bucket: string;
  bucketLabel: string;
  count: number;
  totalOutstanding: number;
};

export type DashboardChartsResult = {
  collectionByProject: CollectionByProjectItem[];
  agingDistribution: AgingDistributionItem[];
};

const BUCKET_LABELS: Record<string, string> = {
  not_due: "غير مستحق",
  "1_30": "1-30 يوم",
  "31_60": "31-60 يوم",
  "61_90": "61-90 يوم",
  "90_plus": "أكثر من 90 يوم",
};

export async function getDashboardCharts(input: {
  projectId?: string;
  sessionUser: SessionUser;
}): Promise<DashboardChartsResult> {
  requirePermission(input.sessionUser, "dashboard.read");

  const data = await loadReadModelData();
  const projectById = buildProjectById(data.projects);
  const visibleContracts = filterContractsByScope(input.sessionUser, data.contracts, input.projectId);
  const installmentsByContract = buildInstallmentsByContract(data.installments);

  // 1. تحصيل حسب المشروع
  const projectTotals = new Map<string, { due: number; collected: number; outstanding: number }>();
  for (const contract of visibleContracts) {
    const installments = installmentsByContract.get(contract.id) ?? [];
    const current = projectTotals.get(contract.project_id) ?? { due: 0, collected: 0, outstanding: 0 };
    for (const inst of installments) {
      current.due += inst.amount_due;
      current.collected += inst.amount_collected;
      current.outstanding += inst.amount_outstanding;
    }
    projectTotals.set(contract.project_id, current);
  }

  const collectionByProject: CollectionByProjectItem[] = Array.from(projectTotals.entries()).map(([projectId, totals]) => ({
    projectName: projectById.get(projectId)?.name_ar ?? projectById.get(projectId)?.name_en ?? projectId,
    totalDue: totals.due,
    totalCollected: totals.collected,
    totalOutstanding: totals.outstanding,
  }));

  // 2. توزيع أعمار المديونية
  const visibleInstallments = visibleContracts.flatMap((c) => installmentsByContract.get(c.id) ?? []);
  const bucketCounts = new Map<string, { count: number; totalOutstanding: number }>();
  for (const inst of visibleInstallments) {
    if (inst.amount_outstanding <= 0) continue;
    const current = bucketCounts.get(inst.delay_bucket) ?? { count: 0, totalOutstanding: 0 };
    current.count += 1;
    current.totalOutstanding += inst.amount_outstanding;
    bucketCounts.set(inst.delay_bucket, current);
  }

  const agingDistribution: AgingDistributionItem[] = ["not_due", "1_30", "31_60", "61_90", "90_plus"]
    .map((bucket) => ({
      bucket,
      bucketLabel: BUCKET_LABELS[bucket] ?? bucket,
      count: bucketCounts.get(bucket)?.count ?? 0,
      totalOutstanding: bucketCounts.get(bucket)?.totalOutstanding ?? 0,
    }))
    .filter((item) => item.count > 0);

  return { collectionByProject, agingDistribution };
}
```

### 8.3 تقرير أعمار المديونية

**ملف جديد**: `src/server/queries/reports/get-aging-report.ts`

```typescript
import type { SessionUser } from "@/lib/auth/get-session-user";
import { requirePermission } from "@/lib/auth/permissions";
import {
  buildCustomerById,
  buildInstallmentsByContract,
  buildProjectById,
  buildUnitCodesByContract,
  filterContractsByScope,
  loadReadModelData,
  paginate,
} from "@/server/queries/read-model-helpers";

const BUCKET_LABELS: Record<string, string> = {
  not_due: "غير مستحق",
  "1_30": "1-30 يوم",
  "31_60": "31-60 يوم",
  "61_90": "61-90 يوم",
  "90_plus": "أكثر من 90 يوم",
};

export type AgingReportRow = {
  customerName: string;
  contractCode: string | null;
  projectName: string;
  unitCodes: string;
  delayBucket: string;
  delayBucketLabel: string;
  delayDays: number;
  amountDue: number;
  amountOutstanding: number;
  dueDate: string;
};

export async function getAgingReport(input: {
  sessionUser: SessionUser;
  projectId?: string;
  bucket?: string;
  page?: number;
  pageSize?: number;
}) {
  requirePermission(input.sessionUser, "reports.read");

  const data = await loadReadModelData();
  const customerById = buildCustomerById(data.customers);
  const projectById = buildProjectById(data.projects);
  const unitCodesByContract = buildUnitCodesByContract(data);
  const installmentsByContract = buildInstallmentsByContract(data.installments);
  const visibleContracts = filterContractsByScope(input.sessionUser, data.contracts, input.projectId);

  const rows: AgingReportRow[] = [];
  for (const contract of visibleContracts) {
    const installments = installmentsByContract.get(contract.id) ?? [];
    for (const inst of installments) {
      if (inst.amount_outstanding <= 0) continue;
      if (input.bucket && inst.delay_bucket !== input.bucket) continue;

      rows.push({
        customerName: customerById.get(contract.customer_id)?.customer_name ?? contract.customer_id,
        contractCode: contract.contract_code,
        projectName: projectById.get(contract.project_id)?.name_ar ?? contract.project_id,
        unitCodes: (unitCodesByContract.get(contract.id) ?? []).join("+"),
        delayBucket: inst.delay_bucket,
        delayBucketLabel: BUCKET_LABELS[inst.delay_bucket] ?? inst.delay_bucket,
        delayDays: inst.delay_days,
        amountDue: inst.amount_due,
        amountOutstanding: inst.amount_outstanding,
        dueDate: inst.due_date,
      });
    }
  }

  rows.sort((a, b) => b.delayDays - a.delayDays);
  const pagination = paginate(rows, input.page, input.pageSize);

  // ملخصات حسب الـ bucket
  const bucketSummary = Object.entries(BUCKET_LABELS).map(([bucket, label]) => {
    const bucketRows = rows.filter((r) => r.delayBucket === bucket);
    return {
      bucket,
      label,
      count: bucketRows.length,
      totalOutstanding: bucketRows.reduce((sum, r) => sum + r.amountOutstanding, 0),
    };
  });

  return {
    ...pagination,
    bucketSummary,
    filters: { projectId: input.projectId ?? null, bucket: input.bucket ?? null },
  };
}
```

## 9. تصدير Excel/CSV/PDF

### 9.1 أداة تصدير Excel

**ملف جديد**: `src/lib/export/export-excel.ts`

```typescript
import ExcelJS from "exceljs";

export type ExportColumn = {
  header: string;
  key: string;
  width?: number;
};

export async function generateExcelBuffer(input: {
  columns: ExportColumn[];
  rows: Record<string, unknown>[];
  sheetName?: string;
  title?: string;
}): Promise<Buffer> {
  const workbook = new ExcelJS.Workbook();
  const sheet = workbook.addWorksheet(input.sheetName ?? "البيانات");

  // RTL
  sheet.views = [{ rightToLeft: true }];

  // عنوان اختياري
  if (input.title) {
    sheet.mergeCells(1, 1, 1, input.columns.length);
    const titleCell = sheet.getCell(1, 1);
    titleCell.value = input.title;
    titleCell.font = { bold: true, size: 14 };
    titleCell.alignment = { horizontal: "center" };
  }

  const headerRowIndex = input.title ? 2 : 1;

  // أعمدة
  sheet.columns = input.columns.map((col) => ({
    header: col.header,
    key: col.key,
    width: col.width ?? 18,
  }));

  // تنسيق Header
  const headerRow = sheet.getRow(headerRowIndex);
  headerRow.values = input.columns.map((col) => col.header);
  headerRow.font = { bold: true };
  headerRow.fill = { type: "pattern", pattern: "solid", fgColor: { argb: "FFE8E8E8" } };

  // بيانات
  for (const row of input.rows) {
    const values = input.columns.map((col) => row[col.key] ?? "");
    sheet.addRow(values);
  }

  return Buffer.from(await workbook.xlsx.writeBuffer());
}
```

### 9.2 أداة تصدير CSV

**ملف جديد**: `src/lib/export/export-csv.ts`

```typescript
export type CsvColumn = {
  header: string;
  key: string;
};

export function generateCsvString(input: {
  columns: CsvColumn[];
  rows: Record<string, unknown>[];
}): string {
  const BOM = "\uFEFF"; // لدعم العربية في Excel
  const headerLine = input.columns.map((col) => escapeCsvField(col.header)).join(",");
  const dataLines = input.rows.map((row) =>
    input.columns.map((col) => escapeCsvField(String(row[col.key] ?? ""))).join(","),
  );
  return BOM + [headerLine, ...dataLines].join("\n");
}

function escapeCsvField(value: string): string {
  if (value.includes(",") || value.includes('"') || value.includes("\n")) {
    return `"${value.replace(/"/g, '""')}"`;
  }
  return value;
}
```

### 9.4 مكون زر التصدير

**ملف جديد**: `src/components/ui/export-button.tsx`

```tsx
"use client";

import { useState } from "react";

type ExportButtonProps = {
  reportType: string;
  filters?: Record<string, string>;
  label?: string;
};

export function ExportButton({ reportType, filters = {}, label = "تصدير Excel" }: ExportButtonProps) {
  const [isExporting, setIsExporting] = useState(false);

  async function handleExport(format: "xlsx" | "csv") {
    setIsExporting(true);
    try {
      const params = new URLSearchParams({ format, ...filters });
      const response = await fetch(`/api/export/${reportType}?${params}`);
      if (!response.ok) throw new Error("فشل التصدير");

      const blob = await response.blob();
      const url = URL.createObjectURL(blob);
      const a = document.createElement("a");
      a.href = url;
      a.download = `${reportType}.${format === "csv" ? "csv" : "xlsx"}`;
      a.click();
      URL.revokeObjectURL(url);
    } catch (error) {
      console.error("Export failed:", error);
    } finally {
      setIsExporting(false);
    }
  }

  return (
    <div className="flex gap-2">
      <button
        className="rounded-2xl border border-border/70 bg-background px-4 py-2 text-sm font-semibold text-foreground disabled:opacity-60"
        disabled={isExporting}
        onClick={() => handleExport("xlsx")}
        type="button"
      >
        {isExporting ? "جاري التصدير..." : label}
      </button>
      <button
        className="rounded-2xl border border-border/70 bg-background px-4 py-2 text-sm font-semibold text-foreground disabled:opacity-60"
        disabled={isExporting}
        onClick={() => handleExport("csv")}
        type="button"
      >
        CSV
      </button>
    </div>
  );
}
```

## التحقق النهائي

1. **تشغيل المشروع محليا**: `npm run dev` — تأكد أن كل الصفحات تعمل بدون أخطاء
2. **تشغيل الاختبارات**: `npm test` — تأكد أن الاختبارات الموجودة (92) لا تزال تمر
3. **التحقق من الرسوم البيانية**: افتح `/dashboard` — يجب أن تظهر الرسوم البيانية
4. **التحقق من التقارير**: افتح `/reports` — يجب أن تظهر 8 تقارير قابلة للفتح
5. **التحقق من التصدير**: اضغط "تصدير Excel" في أي صفحة — يجب أن ينزل ملف Excel
6. **التحقق من تعديل العميل**: افتح ملف عميل → عدّل الملاحظات → احفظ → تأكد من الحفظ
7. **التحقق من تعديل القسط**: افتح صفحة الأقساط → عدّل قسط → تأكد من إعادة حساب الحالة
8. **التحقق من حذف المتابعة**: افتح متابعة → احذفها → تأكد من الحذف
9. **التحقق من اسم المحصل**: افتح المتابعات → يجب أن يظهر اسم المحصل بدل UUID
10. **التحقق من TypeScript**: `npx tsc --noEmit` — لا أخطاء
